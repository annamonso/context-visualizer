% One inter-agent call end to end: the start/done pair, the durable write on
% the producing node, and the forwarded copy flagged so it is never persisted

    msg/.style={->, semithick, black!75},
    ret/.style={->, semithick, black!45, densely dashed},

  ]

  % ---------------------------------------------------------------- lifelines
  \node[life] (a) at (0,0)    {\textbf{agent}\\\texttt{planner}, node $n_1$};
  \node[life] (c) at (3.6,0)  {\textbf{collector}\\node $n_1$};
  \node[life] (k) at (7.1,0)  {\textbf{keeper}\\node $n_1$};
  \node[life] (d) at (10.8,0) {\textbf{dashboard}\\hot store $+$ bus};
  \node[life] (b) at (14.4,0) {\textbf{agent}\\\texttt{executor}, node $n_2$};
